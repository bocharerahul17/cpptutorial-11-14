#include<bits/stdc++.h>
#include<set>
class Solution {
public:
    vector<int> unionArray(vector<int>& nums1, vector<int>& nums2) {

        set<int> st;
        for(auto it: nums1) {
            st.insert(it);
        }

        for(auto it :nums2) {
            st.insert(it);
        }

        vector<int> unionArray;
        for(auto it:st) {
            unionArray.push_back(it);
        }

        return unionArray;
    }
    
};
